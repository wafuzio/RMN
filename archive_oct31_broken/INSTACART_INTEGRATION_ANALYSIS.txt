     "url": "https://...",
     "source_file": "search_results_ice_cream_2025-10-27_14-30-00.html",
     "results": [                          // ❌ Nested structure
       {
         "ads": [
           {
             "type": "Shoppable Display Ad",

             "bbox": {...},
             "advertisers": ["GoodPop"],   // ✅ Already has advertisers array

       }
     ]
   }

   WALMART (Canonical):
   {
     "retailer": "walmart",
     "client": "halo_top",
     "keyword": "ice cream",

     "run_id": "20251027143000",
     "ads": [                               // ✅ Flat structure
       {
         "id": "sba_1",

       }
     ]
   }

   INSTACART:
   - Shoppable Display Ads (image-based carousel ads)

   WALMART:
   - SBA (Sponsored Brand Ads)
   - SBV (Sponsored Brand Video)
   - Tile_Takeover

       return datetime.now(timezone.utc).strftime("%Y%m%d%H%M%S")
   
   def build_run_payload(retailer, client, keyword, run_id):
       return {
           "retailer": retailer,
           "client": client,
           "keyword": keyword,
           "timestamp": now_iso_z(),
           "run_id": run_id,
           "ads": []
       }
   
   def build_ad_object(ad_type, index, advertiser, image_path, ...):
       return {
           "id": f"{ad_type}_{index}",
           "type": ad_type,
           "brand": advertiser or "Unknown",
           "advertisers": [advertiser] if advertiser else [],
           "image_path": image_path,
           ...
